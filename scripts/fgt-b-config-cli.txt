#  Seguridad de Redes · Infraestructura 3 · Maxyohan Montas (2025-0813)
#
#  IMPORTANTE: en el laboratorio TODO se configuró por la GUI (ver el informe).
#  Este archivo es el EQUIVALENTE EN CLI, para reproducir FGT-B desde cero
#  pegándolo en la consola (en un equipo que ya traía la VPN sitio a sitio,
#  primero hay que pasar fgt-b-quitar-sitio-a-sitio-cli.txt).
#
#  Qué publica cada parte:
#    · HTTPS del servidor SIN VPN: Virtual IP 20.25.8.13:443 -> 10.8.13.10:443
#      y la política WAN-to-WEB-HTTPS.
#    · SSH SOLO por la VPN de acceso remoto: túnel dial-up VPN-REMOTO
#      (IKEv1 agresivo + XAuth, Mode Config con el pool 10.25.13.10 - .50) y
#      la política VPN-to-SERVER. No existe ninguna política de SSH desde la WAN.
#
#  Antes de pegarlo sustituye:
#    <CLAVE-PRECOMPARTIDA>  la clave del túnel (la misma que en PC-USER)
#    <CONTRASEÑA-VPNUSER>   la contraseña del usuario de la VPN
#  Ninguna de las dos se publica en el repositorio.
#
#  Bloques (en orden) y su menú en la GUI:
#    config system global ............ System > Settings (zona horaria por CLI)
#    config system interface ......... Network > Interfaces
#    config firewall address ......... Policy & Objects > Addresses
#    config user local / group ....... User & Authentication > User Definition
#                                      y User Groups
#    config vpn ipsec phase1/phase2 .. VPN > IPsec Tunnels > Create New > Custom
#    config firewall vip ............. Policy & Objects > Virtual IPs
#    config router static ............ Network > Static Routes
#    config firewall policy .......... Policy & Objects > Firewall Policy
#
#  Límites de la licencia de evaluación: 3 interfaces (port1, port2 y el túnel
#  VPN-REMOTO), 3 políticas y 3 rutas; cifrado bajo (DES).
#
#  Copia desde «config system global» hasta el final y pégalo en la consola.
#  Las líneas con # de esta cabecera son solo explicación: no las pegues.
# =============================================================================

config system global
    set hostname "FGT-B"
    set timezone 15
    set admintimeout 480
end

config system interface
    edit "port1"
        set alias "LAN-SRV"
        set mode static
        set ip 10.8.13.1 255.255.255.240
        set allowaccess ping https ssh http
    next
    edit "port2"
        set alias "WAN"
        set role wan
        set mode static
        set ip 20.25.8.13 255.255.255.252
        set allowaccess ping
    next
end

config firewall address
    edit "NET-SERVER"
        set subnet 10.8.13.0 255.255.255.240
    next
    edit "VPN-POOL"
        set type iprange
        set start-ip 10.25.13.10
        set end-ip 10.25.13.50
    next
end

config user local
    edit "vpnuser"
        set type password
        set passwd <CONTRASEÑA-VPNUSER>
    next
end

config user group
    edit "VPN-USERS"
        set member "vpnuser"
    next
end

config vpn ipsec phase1-interface
    edit "VPN-REMOTO"
        set type dynamic
        set interface "port2"
        set ike-version 1
        set mode aggressive
        set peertype any
        set net-device disable
        set mode-cfg enable
        set proposal des-sha1
        set dhgrp 2
        set nattraversal enable
        set dpd on-idle
        set keylife 86400
        set xauthtype auto
        set authusrgrp "VPN-USERS"
        set ipv4-start-ip 10.25.13.10
        set ipv4-end-ip 10.25.13.50
        set ipv4-netmask 255.255.255.255
        set psksecret <CLAVE-PRECOMPARTIDA>
    next
end
config vpn ipsec phase2-interface
    edit "VPN-REMOTO"
        set phase1name "VPN-REMOTO"
        set proposal des-sha1
        set pfs enable
        set dhgrp 2
        set replay enable
        set auto-negotiate disable
        set keepalive disable
        set keylifeseconds 43200
        set src-addr-type name
        set dst-addr-type name
        set src-name "NET-SERVER"
        set dst-name "all"
    next
end

config firewall vip
    edit "VIP-WEB-HTTPS"
        set extip 20.25.8.13
        set mappedip "10.8.13.10"
        set extintf "port2"
        set portforward enable
        set extport 443
        set mappedport 443
    next
end

config router static
    edit 1
        set gateway 20.25.8.14
        set device "port2"
    next
end

config firewall policy
    edit 3
        set name "SERVER-to-INTERNET"
        set srcintf "port1"
        set dstintf "port2"
        set action accept
        set srcaddr "NET-SERVER"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set logtraffic all
        set nat enable
    next
    edit 4
        set name "WAN-to-WEB-HTTPS"
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "all"
        set dstaddr "VIP-WEB-HTTPS"
        set schedule "always"
        set service "HTTPS"
        set logtraffic all
    next
    edit 5
        set name "VPN-to-SERVER"
        set srcintf "VPN-REMOTO"
        set dstintf "port1"
        set action accept
        set srcaddr "VPN-POOL"
        set dstaddr "NET-SERVER"
        set schedule "always"
        set service "SSH" "PING" "TRACEROUTE"
        set logtraffic all
    next
end
